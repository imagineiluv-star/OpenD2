cmake_minimum_required(VERSION 3.25)
project(OpenD2Mpq LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(STORM_USE_BUNDLED_LIBRARIES ON CACHE BOOL "" FORCE)
set(STORM_SKIP_INSTALL ON CACHE BOOL "" FORCE)
set(STORM_UNICODE ON CACHE BOOL "" FORCE)
add_subdirectory("${STORMLIB_SOURCE}" stormlib)
add_library(opend2_mpq SHARED mpq.cpp)
target_link_libraries(opend2_mpq PRIVATE storm)
if(WIN32)
 target_compile_definitions(opend2_mpq PRIVATE UNICODE _UNICODE)
endif()
